// Power Query M example for importing the repaired CSV.
// Replace the path below with the path on your computer.
let
    Source = Csv.Document(
        File.Contents("C:\\YOUR\\PATH\\data\\supermarket_sales_cleaned_repaired.csv"),
        [Delimiter=",", Encoding=65001, QuoteStyle=QuoteStyle.Csv]
    ),
    PromotedHeaders = Table.PromoteHeaders(Source, [PromoteAllScalars=true]),
    ChangedDateType = Table.TransformColumnTypes(
        PromotedHeaders,
        {{"Date", type date}},
        "en-US"
    )
in
    ChangedDateType
